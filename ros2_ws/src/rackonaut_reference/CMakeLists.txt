cmake_minimum_required(VERSION 3.8)
project(rackonaut_reference)

find_package(ament_cmake REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  models/reference_drone.sdf config/avoidance-lidar.sdf.fragment config/localization-imu.sdf.fragment config/bridge.yaml)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/models/reference_drone.sdf" drone)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/avoidance-lidar.sdf.fragment" lidar)
# Visibility bit 1 is reserved for this sensor and excludes the host aircraft.
string(REPLACE "version=\"1.8\"" "version=\"1.9\"" drone "${drone}")
string(REGEX REPLACE "(<visual name=\"[^\"]+\">)" "\\1<visibility_flags>4294967293</visibility_flags>" drone "${drone}")
string(REPLACE "      <sensor name=\"rgb\"" "${lidar}\n      <sensor name=\"rgb\"" drone "${drone}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/avoidance_drone.sdf" "${drone}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/localization-imu.sdf.fragment" imu)
string(REPLACE "      <sensor name=\"rgb\"" "${imu}\n      <sensor name=\"rgb\"" localized_drone "${drone}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/localization_drone.sdf" "${localized_drone}")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/bridge.yaml" base_bridge)
string(FIND "${base_bridge}" "- ros_topic_name: /rackonaut/drone/odom" odom_start)
string(FIND "${base_bridge}" "- ros_topic_name: /rackonaut/drone/cmd_vel" command_start)
string(SUBSTRING "${base_bridge}" 0 ${odom_start} common_bridge)
string(SUBSTRING "${base_bridge}" ${command_start} -1 command_bridge)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sensor-navigation.yaml" "${common_bridge}${command_bridge}")
install(DIRECTORY fixtures worlds models config DESTINATION share/${PROJECT_NAME})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/avoidance_drone.sdf" "${CMAKE_CURRENT_BINARY_DIR}/localization_drone.sdf"
  DESTINATION share/${PROJECT_NAME}/models)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/sensor-navigation.yaml" DESTINATION share/${PROJECT_NAME}/config)
install(PROGRAMS scripts/mock_camera DESTINATION lib/${PROJECT_NAME})
ament_package()
