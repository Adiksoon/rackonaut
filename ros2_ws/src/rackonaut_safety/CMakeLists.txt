cmake_minimum_required(VERSION 3.16)
project(rackonaut_safety)

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(ignition-msgs8 REQUIRED)
find_package(ignition-transport11 REQUIRED)

add_executable(infrastructure src/infrastructure.cpp)
target_compile_features(infrastructure PUBLIC cxx_std_17)
ament_target_dependencies(infrastructure rclcpp std_msgs)
target_link_libraries(infrastructure ignition-msgs8::ignition-msgs8 ignition-transport11::core)
install(TARGETS infrastructure DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS src/manage DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
